cmake_minimum_required(VERSION 3.20)
project(M3LlvmOrcJni LANGUAGES C)

set(M3_LLVM_REQUIRED_MAJOR 24 CACHE STRING "Pinned major version of original LLVM fork")
find_package(LLVM CONFIG REQUIRED)
if(NOT LLVM_VERSION_MAJOR EQUAL M3_LLVM_REQUIRED_MAJOR)
  message(FATAL_ERROR "M3 LLVM major mismatch: expected ${M3_LLVM_REQUIRED_MAJOR}, got ${LLVM_VERSION_MAJOR}")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "M3 in-process ORC smoke requires a 64-bit host")
endif()
find_package(JNI REQUIRED)
llvm_map_components_to_libnames(M3_LLVM_LIBS Core OrcJIT NativeCodeGen)
# The opt-in LD_PRELOAD regression must interpose the real LLVM shared ABI;
# do not statically absorb the original ORC C API into this test library.
option(M3_LLVM_ENABLE_FAULT_TEST "Build Linux-only ORC transfer fault interposer" OFF)
if(M3_LLVM_ENABLE_FAULT_TEST AND TARGET LLVM)
  set(M3_LLVM_LIBS LLVM)
endif()
add_library(m3llvmorcjni SHARED src/main/c/m3_llvm_orc_jni.c)
target_include_directories(m3llvmorcjni PRIVATE ${JNI_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_link_libraries(m3llvmorcjni PRIVATE ${M3_LLVM_LIBS})
target_compile_features(m3llvmorcjni PRIVATE c_std_11)
if(MSVC)
  target_compile_options(m3llvmorcjni PRIVATE /W4 /WX)
else()
  target_compile_options(m3llvmorcjni PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(m3llvmorcjni PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Opt-in Linux-only JNI transfer-owner failure regression. Never installed.
if(M3_LLVM_ENABLE_FAULT_TEST)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "M3 LLVM transfer fault injection is Linux-only")
  endif()
  add_library(m3llvmtransferfault SHARED tests/llvm_orc_fault_transfer.c)
  target_include_directories(m3llvmtransferfault PRIVATE ${LLVM_INCLUDE_DIRS})
  target_link_libraries(m3llvmtransferfault PRIVATE ${M3_LLVM_LIBS})
  target_compile_features(m3llvmtransferfault PRIVATE c_std_11)
  if(MSVC)
    message(FATAL_ERROR "M3 LLVM fault interposer requires ELF Linux")
  else()
    target_compile_options(m3llvmtransferfault PRIVATE -Wall -Wextra -Werror)
  endif()
  set_target_properties(m3llvmtransferfault PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
