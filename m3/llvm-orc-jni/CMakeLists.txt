cmake_minimum_required(VERSION 3.20)
project(M3LlvmOrcJni LANGUAGES C)

set(M3_LLVM_REQUIRED_MAJOR 24 CACHE STRING "Pinned major version of original LLVM fork")
find_package(LLVM CONFIG REQUIRED)
if(NOT LLVM_VERSION_MAJOR EQUAL M3_LLVM_REQUIRED_MAJOR)
  message(FATAL_ERROR "M3 LLVM major mismatch: expected ${M3_LLVM_REQUIRED_MAJOR}, got ${LLVM_VERSION_MAJOR}")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "M3 in-process ORC smoke requires a 64-bit host")
endif()
find_package(JNI REQUIRED)
llvm_map_components_to_libnames(M3_LLVM_LIBS Core OrcJIT NativeCodeGen)
add_library(m3llvmorcjni SHARED src/main/c/m3_llvm_orc_jni.c)
target_include_directories(m3llvmorcjni PRIVATE ${JNI_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_link_libraries(m3llvmorcjni PRIVATE ${M3_LLVM_LIBS})
target_compile_features(m3llvmorcjni PRIVATE c_std_11)
if(MSVC)
  target_compile_options(m3llvmorcjni PRIVATE /W4 /WX)
else()
  target_compile_options(m3llvmorcjni PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(m3llvmorcjni PROPERTIES POSITION_INDEPENDENT_CODE ON)
